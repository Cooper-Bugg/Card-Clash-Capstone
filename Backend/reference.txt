How MySQL works, so I don't forget:
- As long as the values follow the order of the specified columns, inserts can be in any order
- Upsert: Default is insert. MySQL only checks primary and unique keys, so if none are included, it assumes an insert.
    If, however, a primary or unique key (like deck_id for decks) is included, MySQL will check for duplicates. If
    there is a duplicate, it will update instead. Be careful of this for teacher table; but for all other tables, the
    only time there should be a primary or unique key in the upsert statement is when a primary key has been pulled from
    the database and is being used to specify an update.

function buildUpsertFields(fields, allowedFields) {
  // 1. Validation & Filtering
  const validFields = Object.keys(fields)
      .filter(key => allowedFields.includes(key) && fields[key] !== undefined);
    if (!validFields.length) {
      return { success: false, error: "No valid fields provided", code: 400 };
    }

    // 2. Build Query
    const allFields = [...validFields];
    const allValues = [...validFields.map(f => fields[f])];

    const columnList = allFields.join(", ");
    const placeholders = allFields.map(() => "?").join(", ");
    const updateClause = allFields.map(f => `${f} = VALUES(${f})`).join(", ");

    return { success: true, allValues, columnList, placeholders, updateClause }
}

function buildConditionFields(conditions, allowedConditionFields) {
  const validConditions = Object.keys(conditions)
    .filter(key => allowedConditionFields.includes(key) && conditions[key] !== undefined);

  if (!validConditions.length) {
    return { success: false, error: "No valid condition fields provided", code: 400 };
  }

  const conditionValues = validConditions.map(f => conditions[f]);
  const whereClause = validConditions.map(f => `${f} = ?`).join(" AND ");

  return { success: true, conditionValues, whereClause };
}

function buildSelectClauses(allowedFields, conditions, allowedConditionFields) {
  const selectClause = allowedFields.length
      ? allowedFields.join(", ")
      : "*";

    // 2. Build WHERE clause from conditions
    const validConditions = Object.keys(conditions)
      .filter(key => allowedConditionFields.includes(key) && conditions[key] !== undefined);

    if (!validConditions.length) {
      return { success: false, error: "No valid conditions provided", code: 400 };
    }

    const whereClause = validConditions.length
      ? `WHERE ${validConditions.map(f => `${f} = ?`).join(" AND ")}`
      : "";

    const values = validConditions.map(f => conditions[f]);

    return { success: true, selectClause, whereClause, values }
}

    const fieldsResponse = buildUpdateFields(fields, allowedFields)

    const conditionsResponse = buildConditionFields(conditions, allowedConditionFields)

/* 
Function used to get records from any table with flexible conditions and field selection. Should
never be exposed directly to the client, but can be used internally by other functions.
Arguments:
- tableName: string, the name of the table to query. Always hardcode in calling functions
- allowedFields: array of strings, fields that are allowed to be selected. Always hardcode
- conditions: object, key-value pairs (JSON) for WHERE clause
- allowedConditionFields: array of strings, fields that are allowed to be used in conditions.
  Always hardcode

Returns:
- On success: { success: true, code: 200, data: [...] }
- On client error (e.g. no valid conditions): { success: false, error: "message", code: 400 }
- On not found: { success: false, error: "No records found", code: 404 }
- On server error: { success: false, error: "message", code: 500 or 503 }
*/
async function getRecords(tableName, allowedFields, conditions = {}, allowedConditionFields = []) {
  try {

    const response = buildSelectClauses(allowedFields, conditions, allowedConditionFields)
    if (!response.success) {
      return response
    }

    const { success, selectClause, whereClause, values } = response

    console.log("[DB Debug] Executing query on table:", tableName);
    // 3. Build and execute query
    const sql = `SELECT ${selectClause} FROM ${tableName} ${whereClause}`;
    const [rows] = await pool.query(sql, values);

    console.log("[DB Debug] Query results:", rows);

    // 4. Handle results
    if (!rows.length) {
      return { success: false, error: "No records found", code: 404 };
    }

    return {
      success: true,
      code: 200,
      data: rows  // mysql2 already returns rows as JSON-friendly objects
    };

  } catch (error) {
    console.error(`[DB Error] Select failed on ${tableName}:`, error.message);
    if (error.code === "ECONNREFUSED") {
      return { success: false, error: "Database unavailable", code: 503 };
    }
    return { success: false, error: "Internal server error", code: 500 };
  }
}

async function getRecordsWithJoins(tableName, allowedFields, conditions = {}, allowedConditionFields = [], joinClauses = []) {
  // Similar to getRecords but also takes an array of JOIN clauses to build more complex queries across multiple tables.
  try {

    for (const join of joinClauses) {
      const ALLOWED_JOIN_TYPES = ["INNER", "LEFT", "RIGHT", "FULL"];
      if (!ALLOWED_JOIN_TYPES.includes(join.type.toUpperCase())) {
        return { success: false, error: `Invalid join type: ${join.type}`, code: 400 };  
      }
    }

    const response = buildSelectClauses(allowedFields, conditions, allowedConditionFields)
    if (!response.success) {
      return response
    }
    const { selectClause, whereClause, values } = response

    // Build JOIN clause

    const joins = []

    for (const join of joinClauses) {
      // Each join should be an object like the following:
      // { type: "INNER", table: "decks", on: "game_sessions.deck_id = decks.deck_id" }
      const { type, table, on } = join;

      const joinClause = ` ${type} JOIN ${table} ON ${on} `;

      joins.push(joinClause);

    }

    // 4. Build and execute query
    let sql = `SELECT ${selectClause} FROM ${tableName}`;
    sql += joins.join(" ");
    sql += ` ${whereClause}`;

    const [rows] = await pool.query(sql, values);

    // 5. Handle results
    if (!rows.length) {
      return { success: false, error: "No records found", code: 404 };
    }

    return {
      success: true,
      code: 200,
      data: rows  // mysql2 already returns rows as JSON-friendly objects
    };

  } catch (error) {
    console.error(`[DB Error] Select failed on ${tableName}:`, error.message);
    if (error.code === "ECONNREFUSED") {
      return { success: false, error: "Database unavailable", code: 503 };
    }
    return { success: false, error: "Internal server error", code: 500 };
  }
